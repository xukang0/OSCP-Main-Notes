## **OSCP LFI checklist**

1. Identify parameters accepting filenames or paths.
2. Test `../../../../etc/passwd`.
3. Test Windows `win.ini` if appropriate.
4. Try additional traversal depth.
5. Try URL encoding and alternative traversal syntax.
6. If successful, enumerate application configuration files.
7. Try `php://filter` to retrieve PHP source.
8. Investigate log poisoning or session poisoning if the conditions exist.
9. Determine whether the vulnerability gives file disclosure only or actual RCE.

## High value files to read
```dataviewjs
const linuxFiles = [
    ["/etc/passwd", "Identify users and home directories"],
    ["/etc/hostname", "Identify hostname"],
    ["/etc/hosts", "Identify configured hosts"],
    ["/var/www/html/config.php", "Potential database credentials"],
    ["/proc/self/cmdline", "Identify process command line"],
    ["/proc/self/environ", "Inspect process environment"]
];

const windowsFiles = [
    ["C:\\Windows\\win.ini", "Confirm file disclosure"],
    ["C:\\Windows\\System32\\drivers\\etc\\hosts", "Inspect host mappings"],
    ["C:\\inetpub\\wwwroot\\web.config", "Potential application credentials"],
    ["Application configuration files", "Find database credentials and secrets"]
];

function createTable(title, files) {
    dv.header(3, title);

    const table = dv.container.createEl("table");
    table.style.width = "100%";
    table.style.borderCollapse = "collapse";

    const header = table.createEl("tr");

    ["File", "Why read it?"].forEach(name => {
        const th = header.createEl("th", { text: name });
        th.style.textAlign = "left";
        th.style.padding = "8px";
    });

    function createCopyButton(cell, value) {
        const text = String(value ?? "").trim();

        const button = cell.createEl("button", { text });

        button.style.cssText = `
            font-family: monospace;
            cursor: pointer;
            padding: 4px 8px;
        `;

        button.addEventListener("click", async () => {
            try {
                await navigator.clipboard.writeText(text);
                button.textContent = "Copied!";
                setTimeout(() => button.textContent = text, 1000);
            } catch {
                button.textContent = "Copy failed";
                setTimeout(() => button.textContent = text, 1000);
            }
        });
    }

    for (const [path, notes] of files) {
        const row = table.createEl("tr");

        const fileCell = row.createEl("td");
        createCopyButton(fileCell, path);

        row.createEl("td", { text: notes });
    }
}

createTable("Linux targets", linuxFiles);
createTable("Windows targets", windowsFiles);
```
